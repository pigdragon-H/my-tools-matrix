## Datamuse API 標準呼叫（必須照此格式）

```typescript
// 標準快取+降級模板（所有Datamuse工具必用）
const CACHE_PREFIX = 'fu_datamuse_'

async function queryDatamuse(endpoint: string): Promise<DatamuseWord[]> {
  const cacheKey = CACHE_PREFIX + endpoint
  const cached = localStorage.getItem(cacheKey)
  if (cached) return JSON.parse(cached)

  try {
    const res = await fetch(`https://api.datamuse.com/words?${endpoint}&md=p&max=30`)
    if (!res.ok) throw new Error(`API ${res.status}`)
    const data = await res.json()
    localStorage.setItem(cacheKey, JSON.stringify(data))
    return data
  } catch {
    return [] // 降級：回傳空陣列，工具顯示友善提示
  }
}

// 各工具對應endpoint：
// Anagram:     rel_anag=${word}
// Rhyme:       rel_rhy=${word}
// Synonym:     rel_syn=${word}
// Antonym:     rel_ant=${word}
// Association: rel_trg=${word}
// Collocation: rel_col=${word}
// Syllable:    sp=${word}&md=s&max=1
```
