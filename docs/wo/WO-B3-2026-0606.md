## Datamuse API 快速參考

```javascript
const BASE = "https://api.datamuse.com/words"

// Anagram（字謎重組）
fetch(`${BASE}?rel_anag=${word}&md=p&max=30`)

// Rhyme（押韻）
fetch(`${BASE}?rel_rhy=${word}&md=p&max=30`)

// Association（聯想）
fetch(`${BASE}?rel_trg=${word}&md=p&max=20`)

// Synonym（同義詞）
fetch(`${BASE}?rel_syn=${word}&md=p&max=20`)

// Antonym（反義詞）
fetch(`${BASE}?rel_ant=${word}&md=p&max=20`)

// Collocation（搭配詞）
fetch(`${BASE}?rel_col=${word}&md=p&max=20`)

// Syllable count（音節數）
fetch(`${BASE}?sp=${word}&md=s&max=1`)
// 回傳：[{ word, numSyllables: 2 }]

// 回傳格式（通用）：
// [{ word: "listen", score: 2906, tags: ["v"], numSyllables: 2 }]
// tags: n=名詞 v=動詞 adj=形容詞 adv=副詞
```
